// 这是一个思源笔记插件的初始化示例代码
import { Plugin } from "siyuan";

export default class StyleEditorPlugin extends Plugin {
    async onload() {
        console.log("Style Editor Plugin loaded successfully!");
        this.registerCommand();
    }

    registerCommand() {
        this.addCommand({
            langKey: "showStyleEditor",
            hotkey: "⌥⌘S",
            callback: () => {
                this.showEditorDialog();
            }
        });
    }

    showEditorDialog() {
        // 打开样式编辑器弹窗逻辑
    }
}
