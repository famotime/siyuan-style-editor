// 右侧工具栏鼠标位置跟随滚动
// version 3.0 - 移除标签融合，仅保留工具栏滚动功能
(() => {
    const rightToolBarWidth = 400;

    Math.clamp || (Math.clamp = function(value, min, max) {
        return Math.min(Math.max(value, min), max);
    });
  
    function whenExist(selector, func_cb, time_out = 4000) {
        return new Promise((resolve) => {
            const startTime = Date.now();
            const checkForElement = () => {
                let element = null;
                if (typeof selector === 'string') {
                    if (selector.startsWith('#')) {
                        element = document.getElementById(selector.slice(1));
                    } else {
                        element = document.querySelector(selector);
                    }
                } else if (typeof selector === 'function') {
                    element = selector();
                } else {
                    console.error("Invalid selector type");
                    resolve(false);
                    return;
                }

                if (element) {
                    if (func_cb) func_cb(element);
                    resolve(true);
                } else if (Date.now() - startTime >= time_out) {
                    console.log(selector, "whenExist timeout");
                    resolve(false);
                } else {
                    requestAnimationFrame(checkForElement);
                }
            };
            checkForElement();
        });
    }

    let styleElement = null;

    function set_css_style() {
        if (styleElement) {
            styleElement.remove();
            styleElement = null;
        }
        styleElement = document.createElement('style');
        styleElement.id = 'ToolbarScroll';
        styleElement.textContent = `

    // 创建滚动容器并组织工具栏布局
    function setupToolbarLayout() {
        const toolbar = document.getElementById('toolbar');
        if (!toolbar) {
            console.log('[Toolbar] Toolbar not found');
            return;
        }

        // 如果已经存在滚动容器，则不重复创建
        let scrollContainer = toolbar.querySelector('.toolbar-scroll-wrap');
        if (scrollContainer) {
            console.log('[Toolbar] Scroll container already exists');
            return;
        }

        scrollContainer = document.createElement('div');
        scrollContainer.className = 'toolbar-scroll-wrap';
        scrollContainer.id = 'toolbar-scroll-wrap';
    
        const windowControls = document.getElementById('windowControls');
        if (windowControls) {
            toolbar.insertBefore(scrollContainer, windowControls);
        } else {
            toolbar.appendChild(scrollContainer);
        }
        console.log('[Toolbar] Scroll container created');

        // 移动元素到滚动容器
        const allItems = Array.from(toolbar.children);
        let movedCount = 0;
    
        allItems.forEach(item => {
            // 跳过不需要放入滚动容器的元素
            if (item.id === 'barWorkspace' || 
                item.id === 'drag' || 
                item.id === 'windowControls' ||
                item.classList.contains('toolbar-scroll-wrap') ||
                item.closest('#windowControls')) {
                return;
            }
        
            if (item.parentElement !== scrollContainer) {
                scrollContainer.appendChild(item);
                movedCount++;
            }
        });
    
        console.log(`[Toolbar] Moved ${movedCount} items to scroll container`);

        // 确保窗口控制按钮在最后
        if (windowControls && windowControls.parentElement !== toolbar) {
            toolbar.appendChild(windowControls);
        }

        // 初始化 version 1.3 的鼠标位置跟随滚动
        initVersion13Scroll(scrollContainer);
    }

    // Version 1.3 的鼠标位置跟随滚动（唯一实现的滚动模式）
    function initVersion13Scroll(element) {
        if (!element) {
            console.log('[Scroll] Container is null');
            return;
        }

        console.log('[Scroll] Initializing version 1.3 mouse position follow scroll');

        // 计算最大子元素宽度
        let maxChildWidth = 23.5;
        [...element.children].forEach(el => {
            const rect = el.getBoundingClientRect();
            if(rect.width > maxChildWidth) maxChildWidth = rect.width;
        });

        // 监听鼠标移动事件，根据鼠标位置滚动
        element.addEventListener("mousemove", function(mouseEvent) {
            var visibleWidth = element.clientWidth;
            var totalScrollWidth = element.scrollWidth;

            // 只有当内容溢出时才启用滚动
            if (totalScrollWidth > visibleWidth) {
                // 计算鼠标在元素内的相对 X 坐标（减去元素左边缘和最大子元素宽度）
                var elementRect = element.getBoundingClientRect();
                var adjustedMouseX = mouseEvent.clientX - elementRect.left - maxChildWidth;
            
                // 计算滚动百分比
                var scrollPercentage = adjustedMouseX / (visibleWidth - maxChildWidth);

                // 钳位到 [0, 1] 范围，防止越界
                scrollPercentage = Math.clamp(scrollPercentage, 0, 1);

                // 设置水平滚动位置
                element.scrollLeft = (totalScrollWidth - visibleWidth) * scrollPercentage;
            }
        });

        console.log('[Scroll] Version 1.3 scroll mode enabled');
    }

    // 监听工具栏变化，动态调整滚动容器
    function observeToolbarChanges() {
        whenExist('#toolbar', (toolbar) => {
            const toolbarObserver = new MutationObserver(() => {
                setTimeout(setupToolbarLayout, 100);
            });
        
            toolbarObserver.observe(toolbar, { 
                childList: true, 
                subtree: false
            });
        
            console.log('[Toolbar] Observing toolbar changes');
        });
    }

    // 初始化
    set_css_style();
  
    // 延迟初始化，确保 DOM 已加载
    setTimeout(() => {
        setupToolbarLayout();
    }, 500);
  
    setTimeout(() => {
        setupToolbarLayout();
    }, 1000);
  
    setTimeout(() => {
        setupToolbarLayout();
    }, 2000);

    // 监听工具栏变化
    observeToolbarChanges();
})();

    function initTabBarScroll(element, index) {
        if (!element) {

                    handleScroll(mouseEvent);
                    ticking = false;

                ticking = true;
